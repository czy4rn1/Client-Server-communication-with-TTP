72 def encrypted_msg(self, message):
73
74 iv = os.urandom(16)
75 padder = aes_padding.PKCS7(128).padder()
76 padded_data = padder.update(message) + padder.finalize()
77 cipher = Cipher(algorithms.AES(self.session_key), modes.CBC(iv))
78 encryptor = cipher.encryptor()
79 text = encryptor.update(padded_data) + encryptor.finalize()
80 return iv + text

84 def decrypt_msg(self, data):
85 iv = data[:16]
86 text = data[16:]
87 cipher = Cipher(algorithms.AES(self.session_key), modes.CBC(iv))
88 decryptor = cipher.decryptor()
89 padded_data = decryptor.update(text) + decryptor.finalize()
90 unpadder = aes_padding.PKCS7(128).unpadder()
91 try:
92 final_data = unpadder.update(padded_data) + unpadder.finalize()
93 return final_data.decode('utf-8')
94 except ValueError:
95 return padded_data.decode('utf-8', errors='ignore')

98 def encrypt_id_and_obtain_x509cert(self):
99 global logged_in_ttp
100 ttp_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
101 ttp_socket.connect((ttp_host, ttp_port))
102 ttp_socket.sendall(b'client_login')
103 logger.info(' User has logged in to TTP')
104 ttp_key_length = receive_data(ttp_socket, 4)
105 ttp_key_length = int.from_bytes(ttp_key_length, byteorder='big')
106 ttp_key = b''
107 while len(ttp_key) < ttp_key_length:
108 data = ttp_socket.recv(ttp_key_length - len(ttp_key))
109 if not data: break
110 ttp_key += data
111
112 ttp_public_key = serialization.load_pem_public_key(ttp_key)
113 logger.info(' User has received TTP Public Key')
114
115 self.encryptedID = ttp_public_key.encrypt(self.clientID,
116 padding.OAEP(mgf=padding.MGF1(algorithm=hashes.SHA256()),
117 algorithm=hashes.SHA256(),
118 label=None))
119 logger.info(' Encrypted my ID with TTP Public Key')
120 ttp_socket.sendall(b'X509_REQUEST')
121 encID_len = len(self.encryptedID)
122 ttp_socket.sendall(encID_len.to_bytes(4, byteorder='big'))
123 ttp_socket.sendall(self.encryptedID)
124 logger.info(' User has sent their encrypted ID to TTP')
125
126 public_pem = self.public_key.public_bytes(encoding=serialization.Encoding.PEM,
127 format=serialization.PublicFormat.SubjectPublicKeyInfo)
128 ttp_socket.sendall(len(public_pem).to_bytes(4, byteorder='big'))
129 ttp_socket.sendall(public_pem)
130 logger.info(' User has sent their Public Key to TTP')
131
132 raw_cert_len = receive_data(ttp_socket, 4)
133 if raw_cert_len:
134 cert_len = int.from_bytes(raw_cert_len, byteorder='big')
135 cert_pem = receive_data(ttp_socket, cert_len)
136 self.x509cert = x509.load_pem_x509_certificate(cert_pem)
137 logger.info(' User has obtained x509 certificate')
138 ttp_socket.close()
139 ttp_socket = None
140 logged_in_ttp = True
