197 def handle(self, conn):
198 try:
199 raw_id_len = receive_data(conn, 4)
200 if not raw_id_len: return
201 id_len = int.from_bytes(raw_id_len, byteorder='big')
202 encID = receive_data(conn, id_len)
203 logger.info(' Received encrypted ID')
204 raw_pub_len = receive_data(conn, 4)
205 if not raw_pub_len: return
206 pub_len = int.from_bytes(raw_pub_len, byteorder='big')
207 client_pub_pem = receive_data(conn, pub_len)
208 logger.info(' Received Public Key')
209
210 encID_bytes = self.private_key.decrypt(encID, padding.OAEP(
211 mgf=padding.MGF1(algorithm=hashes.SHA256()),
212 algorithm=hashes.SHA256(),
213 label=None
214 ))
215 client_id_hex = encID_bytes.hex()
216 client_pub_key = serialization.load_pem_public_key(client_pub_pem)
217 cert = self.create_x509(client_id_hex, client_pub_key)
218 logger.info(' Created and signed clients x509 certificate')
219 cert_pem = cert.public_bytes(serialization.Encoding.PEM)
220 conn.sendall(len(cert_pem).to_bytes(4, byteorder='big'))
221 conn.sendall(cert_pem)
222 logger.info(' Sent x509 certificate')
223 except Exception as e:
224 conn.sendall(b'ERROR')
225 print(e)

230 def create_x509(self, client_id_hex, client_pub_key):
231 client = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, client_id_hex)])
232 issuer = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, "TTP")])
233
234 cert = x509.CertificateBuilder().subject_name(client).issuer_name(issuer).public_key(client_pub_key). \
235 serial_number(x509.random_serial_number()).not_valid_before(datetime.datetime.utcnow()). \
236 not_valid_after(datetime.datetime.utcnow() + datetime.timedelta(days=2)). \
237 sign(self.private_key, hashes.SHA256())
238 return cert
