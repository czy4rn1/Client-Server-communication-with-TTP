69 def encrypted_msg(self, message):
70
71 iv = os.urandom(16)
72 padder = aes_padding.PKCS7(128).padder()
73 padded_data = padder.update(message) + padder.finalize()
74 cipher = Cipher(algorithms.AES(self.session_key), modes.CBC(iv))
75 encryptor = cipher.encryptor()
76 text = encryptor.update(padded_data) + encryptor.finalize()
77 return iv + text

81 def decrypt_msg(self, data):
82 iv = data[:16]
83 text = data[16:]
84 cipher = Cipher(algorithms.AES(self.session_key), modes.CBC(iv))
85 decryptor = cipher.decryptor()
86 padded_data = decryptor.update(text) + decryptor.finalize()
87 unpadder = aes_padding.PKCS7(128).unpadder()
88 try:
89 final_data = unpadder.update(padded_data) + unpadder.finalize()
90 return final_data.decode('utf-8')
91 except ValueError:
92 return padded_data.decode('utf-8', errors='ignore')

95 def encrypt_id_and_obtain_x509cert(self):
96 global logged_in_ttp
97 ttp_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
98 ttp_socket.connect((ttp_host, ttp_port))
99 ttp_socket.sendall(b'client_login')
100 logger.info(' Logged in to TTP')
101 ttp_key_length = receive_data(ttp_socket, 4)
102 ttp_key_length = int.from_bytes(ttp_key_length, byteorder='big')
103 ttp_key = b''
104 while len(ttp_key) < ttp_key_length:
105 data = ttp_socket.recv(ttp_key_length - len(ttp_key))
106 if not data: break
107 ttp_key += data
108
109 ttp_public_key = serialization.load_pem_public_key(ttp_key)
110 logger.info(' Received TTP Public Key')
111
112 self.encryptedID = ttp_public_key.encrypt(self.clientID,
113 padding.OAEP(mgf=padding.MGF1(algorithm=hashes.SHA256()),
114 algorithm=hashes.SHA256(),
115 label=None))
116 logger.info(' Encrypted my ID with TTP Public Key')
117 ttp_socket.sendall(b'X509_REQUEST')
118 encID_len = len(self.encryptedID)
119 ttp_socket.sendall(encID_len.to_bytes(4, byteorder='big'))
120 ttp_socket.sendall(self.encryptedID)
121 logger.info(' Sent my encrypted ID to TTP')
122
123 public_pem = self.public_key.public_bytes(encoding=serialization.Encoding.PEM,
124 format=serialization.PublicFormat.SubjectPublicKeyInfo)
125 ttp_socket.sendall(len(public_pem).to_bytes(4, byteorder='big'))
126 ttp_socket.sendall(public_pem)
127 logger.info(' Sent my Public Key to TTP')
128
129 raw_cert_len = receive_data(ttp_socket, 4)
130 if raw_cert_len:
131 cert_len = int.from_bytes(raw_cert_len, byteorder='big')
132 cert_pem = receive_data(ttp_socket, cert_len)
133 self.x509cert = x509.load_pem_x509_certificate(cert_pem)
134 logger.info(' Obtained x509 certificate')
135 ttp_socket.close()
136 ttp_socket = None
137 logged_in_ttp = True
